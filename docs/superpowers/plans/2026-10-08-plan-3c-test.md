# Circuit Map Plan 3c: The Breaker Test Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** From a breaker's page he can run a guided test: switch the breaker off, walk the house marking what has no power, switch it back on, and have the app record what the breaker feeds, correcting wrong guesses on the way. A bar saying the breaker is OFF stays on every screen, and survives closing the app, until he says it is back on.

**Architecture:** The test in progress is one plain object, the *session* (`app.test`): which breaker, whether it is off, what was marked dead (with what each device said before, so a mark can be taken back), which floors he walked. It is kept on the device with `store.setSetting("test", ...)`, not in the project file. Marking a device writes its breaker to the project at once through `app.commit`, so Undo works; the session is tidied against the project on every draw. The screens are one more page of the Breakers tab (`view/test.js`), in four stages: steps, marking, "still assigned here", summary. All rules and wording are in `src/testing.js` with tests.

**Tech Stack:** As before: plain ES modules, `node --test`, no dependencies.

**Spec:** `docs/superpowers/specs/2026-10-06-circuit-map-app-design.md` (Screens > Test; Using it: "One banner at the top at a time").

**Conventions for every task**

- Work on the branch `plan-3c-test`.
- Tests: `node --test`. Try it: `node tools/serve.mjs`, then http://localhost:8765.
- Views change the project only through `app.commit(...)`. After anything that changes the session, call `app.saveTest()`.
- The screens say "test", never "sweep"; "No power", with a cross and the words, never only a colour.
- Commit messages end with the line `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.
- Where a step says "Replace A with B", A appears exactly once in the file.

  assert.deepEqual([joinLabels([]), joinLabels(["A"]), joinLabels(["A", "B"]), joinLabels(["A", "B", "C"])], ["", "A", "A and B", "A, B and C"]);

test("the header names the breaker and says how far the testing has got", () => {
  const { p, main, kit, b1, b2, panel } = house();
  const head = testHeader(p, b1);
  assert.deepEqual([head.title, head.breakers, head.devices], ["Breaker 1, Kitchen, 15 A", "Breakers tested: 0 of 3", "0 of 4 devices have a breaker"]);

  assert.deepEqual(nextToTest(p).label, "");            // breaker 3; breaker 2 feeds a subpanel

  m.placeDevice(p, { floorId: main.id, roomId: kit.id, type: "D", x: 1, y: 1 }, 3000);
  assert.equal(progressText(p), "Breakers to test: 2. New devices added.");

```js
// The rules of a breaker test. A test in progress is a session: a plain object kept on this device (not in the
// project file), so the "breaker is OFF" bar and what he has marked survive closing the app.
import { find, setBreaker, unsweptBreakers, unconfirmedOnBreaker, protectedBy } from "./model.js";
import { typeInfo, isSwitch } from "./catalog.js";
import { plural } from "./flow.js";
import { breakerName } from "./panel.js";

// dead: { deviceId: { breakerId, tested } } is what each marked device said before, so a mark can be taken back.
// floors: the floors he ticked as walked. kept: devices he chose to leave on this breaker in the last step.
export function newSession(breakerId) {
  return { breakerId, off: false, steps: [false, false], view: "list", dead: {}, floors: [], kept: [], review: false, done: false, summary: null };
}

export function joinLabels(labels) {
  return labels.length < 2 ? labels.join("") : `${labels.slice(0, -1).join(", ")} and ${labels.at(-1)}`;
}

// Things that go dead when a breaker is off, and are not already known to be dead for another reason.
const testable = (d) => typeInfo(d.type).sweep && !d.notWorking;

// Breakers that get tested: not an empty space, and not one that feeds a subpanel (its devices belong to the
// subpanel's own breakers).
const tested = (project) => project.breakers.filter((b) => b.kind !== "empty" && !project.panels.some((p) => p.fedFromBreakerId === b.id));
const toTest = (project) => {
  const ids = new Set(tested(project).map((b) => b.id));
  return unsweptBreakers(project).filter((b) => ids.has(b.id));
};

// "Breakers tested: 5 of 18", or, when devices added since make tested breakers count again,
// "Breakers to test: 18. New devices added."
export function progressText(project) {
  const all = tested(project), left = toTest(project);
  return left.some((b) => b.swept) ? `Breakers to test: ${left.length}. New devices added.` : `Breakers tested: ${all.length - left.length} of ${all.length}`;
}

export function testHeader(project, breaker) {
  const devices = project.devices.filter((d) => typeInfo(d.type).sweep);
  return {
    title: [`Breaker ${breakerName(breaker)}`, breaker.label, `${breaker.amps} A`].filter(Boolean).join(", "),
    breakers: progressText(project),
    devices: `${devices.filter((d) => d.breakerId).length} of ${devices.length} devices have a breaker`,
  };
}

// The next breaker to test, in panel then space order; null when none is left.
export function nextToTest(project, afterId = "") {
  return toTest(project).find((b) => b.id !== afterId) || null;
}

// For steps 1 and 2: how many switches and GFCI outlets the app knows of.
export function stepCounts(project) {
  const count = (keep) => project.devices.filter(keep).length;
  const text = (n) => (n ? `${n} recorded` : "None recorded");
  const switches = count((d) => isSwitch(d.type)), gfcis = count((d) => d.type === "G");
  return { switches, gfcis, switchText: text(switches), gfciText: text(gfcis) };
}

// Is there anything to mark at all?
export function canTest(project) {
  return project.devices.some(testable);
}

// An earlier test found this device on another breaker. (A device only guessed onto another breaker simply moves.)
export function hasConflict(project, device, session) {
  return device.tested && device.breakerId !== session.breakerId && !!find(project, "breakers", device.breakerId);
}

export function conflictText(project, device, session) {
  const other = find(project, "breakers", device.breakerId), now = find(project, "breakers", session.breakerId);
  return `${device.label} was dead when breaker ${breakerName(other)} was off too. Move it to breaker ${breakerName(now)}?`;
}

// "No power": the device is on this breaker, found by a test.
export function markDead(project, session, deviceId, now = Date.now()) {
  const d = find(project, "devices", deviceId);
  if (!d || session.dead[deviceId]) return;
  session.dead[deviceId] = { breakerId: d.breakerId, tested: d.tested };
  setBreaker(project, deviceId, session.breakerId, true, now);
}

// Tapped again: it goes back to what it said before (or to nothing, if that breaker has gone).
export function unmark(project, session, deviceId, now = Date.now()) {
  const before = session.dead[deviceId];
  if (!before) return;
  delete session.dead[deviceId];
  const back = find(project, "breakers", before.breakerId) ? before : { breakerId: "", tested: false };
  setBreaker(project, deviceId, back.breakerId, back.tested, now);
}

// Drop marks the project no longer agrees with (Undo took them back, or the device was deleted).
export function reconcile(project, session) {
  for (const id of Object.keys(session.dead)) {
    const d = find(project, "devices", id);
    if (!d || d.breakerId !== session.breakerId || !d.tested) delete session.dead[id];
  }
}

const canMark = (project, session) => (d) => !session.dead[d.id] && testable(d) && !hasConflict(project, d, session);

// "Also mark the 5 outlets it protects?": the ids that would be marked.
export function alsoProtected(project, session, gfci) {
  return protectedBy(project, gfci.id).filter(canMark(project, session)).map((d) => d.id);
}

// "Whole room has no power": the ids that would be marked. group is one entry of model.sweepList.
export function roomToMark(project, session, group) {
  return group.devices.filter(canMark(project, session)).map((d) => d.id);
}

// After the breaker is back on: devices on the floors he walked that are on this breaker but were not marked.
// { guessed, tested, text }: "KIT-05 and KIT-07 were put on breaker 12 but still had power."
export function stillAssigned(project, session) {
  const found = unconfirmedOnBreaker(project, session.breakerId, Object.keys(session.dead), session.floors);
  const keep = (d) => testable(d) && !session.kept.includes(d.id);
  const guessed = found.guessed.filter(keep), tested = found.tested.filter(keep), all = [...guessed, ...tested];
  const breaker = find(project, "breakers", session.breakerId);
  return {
    guessed, tested,
    text: all.length ? `${joinLabels(all.map((d) => d.label))} ${all.length === 1 ? "was" : "were"} put on breaker ${breakerName(breaker)} but still had power.` : "",
  };
}

// { found, guesses, left }: what the finished test says.
export function finishSummary(project, session) {
  const breaker = find(project, "breakers", session.breakerId);
  const devices = Object.keys(session.dead).map((id) => find(project, "devices", id)).filter(Boolean)
    .sort((a, b) => a.label.localeCompare(b.label, "en", { numeric: true }));
  const guesses = devices.map((d) => {
    const before = session.dead[d.id], was = find(project, "breakers", before.breakerId);
    return was && !before.tested && was.id !== breaker.id ? `${d.label} was your guess of breaker ${breakerName(was)}` : "";
  }).filter(Boolean);
  return {
    found: devices.length
      ? `${plural(devices.length, "device")} found on breaker ${breakerName(breaker)}: ${devices.map((d) => d.label).join(", ")}`
      : `Nothing was found on breaker ${breakerName(breaker)}.`,
    guesses,
    left: toTest(project).filter((b) => b.id !== breaker.id).length,
  };
}
```

// The breaker test: one page of the Breakers tab, in four stages.
//   steps   - turn switches on, reset GFCIs, switch the breaker off
//   marking - walk the house and mark what has no power (a list, or the plan)
//   review  - "still assigned here": things on this breaker that had power
//   summary - what was found, and the next breaker
// The session (app.test) says which stage it is.
import * as m from "../model.js";
import { displayName, typeInfo } from "../catalog.js";
import { markersNear } from "../placing.js";
import { plural } from "../flow.js";
import { breakerName, breakerTitle } from "../panel.js";
import { newSession, joinLabels, testHeader, stepCounts, canTest, hasConflict, conflictText, markDead, unmark, reconcile,
  alsoProtected, roomToMark, stillAssigned, finishSummary, nextToTest } from "../testing.js";
import { h, button, ask, toast } from "../ui.js";
import { floorSwitch } from "./floorswitch.js";
import { drawFloor } from "./markers.js";
import { state, show } from "./panel.js";

const REACH = 22; // screen pixels: half of a 44-point target

export function startTest(app, breaker) {
  app.test = newSession(breaker.id);
  app.saveTest();
  show(app, { page: "test", breakerId: breaker.id });
}

// A change to the project made during the test, then the session saved with it.
const change = (app, name, fn) => {
  app.commit(name, fn);
  app.saveTest();
};

// A change to the session only.
const redraw = (app) => {
  app.saveTest();
  app.render();
};

const drop = (app) => {
  app.test = null;
  app.saveTest();
};

export function testPage(app) {
  const p = app.project, t = app.test, breaker = m.find(p, "breakers", t.breakerId);
  reconcile(p, t);
  const head = testHeader(p, breaker);
  const top = [
    // Leaving before the breaker is off abandons the test. Once it is off the bar keeps it alive.
    button("Back to breakers", () => {
      if (!t.off) drop(app);
      show(app, { page: "" });
    }),
    h("h2", {}, head.title),
    h("p", { class: "small" }, head.breakers),
    h("p", { class: "small" }, head.devices),
  ];
  if (t.done) return summaryPage(app, breaker, top);
  if (t.review) return reviewPage(app, breaker, top);
  if (!canTest(p)) {
    return h("section", { class: "page" }, top,
      h("p", {}, "No outlets or lights placed yet. Add them on the Add tab first."),
      button("Go to Add", () => { drop(app); app.go("add"); }, "main"));
  }
  return t.off ? markPage(app, breaker, top) : stepsPage(app, breaker, top);
}

// ---- stage 1: the three steps

function stepsPage(app, breaker, top) {
  const t = app.test, counts = stepCounts(app.project);
  // With none recorded a step counts as done.
  const step = (i, text, why, none) => h("button", {
    type: "button", class: "step", "aria-pressed": String(t.steps[i] || none), disabled: none,
    onclick: () => { t.steps[i] = !t.steps[i]; redraw(app); },
  }, h("strong", {}, text), h("span", { class: "small" }, why));
  const ready = (t.steps[0] || !counts.switches) && (t.steps[1] || !counts.gfcis);
  return h("section", { class: "page" }, top,
    step(0, "1. Turn every light switch on", `A switched light looks dead even when its breaker is on. ${counts.switchText}.`, !counts.switches),
    step(1, "2. Press Reset on every GFCI outlet", `A tripped one would be mistaken for a dead circuit, and it cannot be reset once the breaker is off. ${counts.gfciText}.`, !counts.gfcis),
    h("button", { type: "button", class: "step main", disabled: !ready, onclick: () => { t.off = true; redraw(app); } },
      h("strong", {}, `3. Switch breaker ${breakerName(breaker)} off, then tap here`),
      h("span", {}, "Keep it short if a freezer or fridge is on this breaker.")),
    !ready && h("p", { class: "small" }, "Tick steps 1 and 2 first."));
}

// ---- stage 2: marking what has no power

function markPage(app, breaker, top) {
  const p = app.project, t = app.test, groups = m.sweepList(p, breaker.id);
  const pick = (value, text) => h("button", { type: "button", "aria-pressed": String(t.view === value), onclick: () => { t.view = value; redraw(app); } }, text);
  // Only floors he ticks count in the last step: a device on a floor he has not walked has not been seen.
  const checked = (floor) => h("button", {
    type: "button", "aria-pressed": String(t.floors.includes(floor.id)),
    onclick: () => {
      t.floors = t.floors.includes(floor.id) ? t.floors.filter((id) => id !== floor.id) : [...t.floors, floor.id];
      redraw(app);
    },
  }, "I've checked this floor");
  const head = [top,
    h("p", {}, `Breaker ${breakerName(breaker)} is off. Mark everything that has no power.`),
    h("div", { class: "pick-row", role: "group", "aria-label": "Show as" }, pick("list", "List"), pick("plan", "Plan"))];
  const foot = button("I've switched it back on", () => backOn(app), "main big");

  if (t.view === "plan") {
    const floor = app.floor();
    const devices = groups.filter((g) => g.floorId === floor.id).flatMap((g) => g.devices);
    let plan;
    if (floor.imageId) {
      app.showPicture(floor);
      drawFloor(app, floor, { only: new Set(devices.map((d) => d.id)), ringed: new Set(Object.keys(t.dead)) });
      app.plan.onTap = ({ x, y }) => {
        const d = markersNear(devices, x, y, REACH / app.plan.view.k)[0];
        if (d) toggleDead(app, d.id);
      };
      plan = app.plan.el;
    } else {
      plan = h("p", { class: "plan-hint" }, "This floor has no picture. Use the list.");
    }
    return h("section", { class: "plan-screen" },
      h("div", { class: "page" }, head),
      floorSwitch(app),
      h("div", { class: "pick-row" }, checked(floor), h("span", {}, `${devices.filter((d) => t.dead[d.id]).length} marked No power here`)),
      plan,
      h("div", { class: "page" }, foot));
  }

  const row = (d) => {
    const dead = !!t.dead[d.id];
    const other = !dead && d.breakerId !== breaker.id && m.find(p, "breakers", d.breakerId);
    return h("li", {}, h("button", { type: "button", class: "row dead-toggle", "aria-pressed": String(dead), onclick: () => toggleDead(app, d.id) },
      h("strong", {}, `${d.label}, ${displayName(d)}`),
      h("span", {}, dead ? "✗ No power" : "Tap if it has no power"),
      other && h("span", { class: "small" }, `Now on breaker ${breakerName(other)}${d.tested ? "" : " (guess)"}`)));
  };
  const notWorking = m.notWorking(p).filter((d) => typeInfo(d.type).sweep);
  return h("section", { class: "page" }, head,
    m.sortedFloors(p).map((floor) => {
      const here = groups.filter((g) => g.floorId === floor.id);
      return here.length > 0 && [
        h("div", { class: "buttons" }, h("h3", {}, floor.name), checked(floor)),
        here.map((g) => [
          h("div", { class: "buttons" },
            h("strong", {}, g.roomName),
            roomToMark(p, t, g).length > 1 && button("Whole room has no power", () => wholeRoom(app, g))),
          h("ul", { class: "rows" }, g.devices.map(row)),
        ]),
      ];
    }),
    // Left out of the list until cleared, so a dead bulb is not mistaken for a dead circuit.
    notWorking.length > 0 && h("details", {},
      h("summary", {}, `Not working (${notWorking.length})`),
      h("p", { class: "small" }, "Fix them or find out why; they cannot be tested until then."),
      h("ul", { class: "rows" }, notWorking.map((d) => h("li", {},
        h("span", {}, `${d.label}, ${displayName(d)}`),
        button("Working now", () => change(app, `marked ${d.label} as working`, (p) => m.setNotWorking(p, d.id, false))))))),
    foot);
}

// Tap a device: "No power". Tap again: unmarked.
function toggleDead(app, id) {
  const t = app.test, d = m.find(app.project, "devices", id), breaker = m.find(app.project, "breakers", t.breakerId);
  if (t.dead[id]) {
    change(app, `unmarked ${d.label}`, (p) => unmark(p, t, id));
    return;
  }
  const mark = () => {
    change(app, `found ${d.label} on ${breakerTitle(breaker)}`, (p) => markDead(p, t, id));
    // A GFCI outlet that is dead takes what it protects with it.
    const also = d.type === "G" ? alsoProtected(app.project, t, d) : [];
    if (!also.length) return;
    ask(`Also mark the ${plural(also.length, "device")} it protects?`, "", [
      { text: "Yes", cls: "main", run: () => change(app, `found ${plural(also.length, "device")} behind ${d.label} on ${breakerTitle(breaker)}`, (p) => { for (const x of also) markDead(p, t, x); }) },
      { text: "No" },
    ]);
  };
  if (!hasConflict(app.project, d, t)) return mark();
  const other = m.find(app.project, "breakers", d.breakerId);
  ask(`${d.label} was found before`, conflictText(app.project, d, t), [
    { text: "Move it", cls: "main", run: mark },
    { text: `Keep it on breaker ${breakerName(other)}` },
    // For an outlet that goes dead whatever breaker is off: a tripped GFCI, a loose wire, a switch that is off.
    { text: "It has no power for another reason", cls: "plain", run: () => change(app, `marked ${d.label} as not working`, (p) => m.markNoBreaker(p, id)) },
  ]);
}

function wholeRoom(app, group) {
  const t = app.test, ids = roomToMark(app.project, t, group), breaker = m.find(app.project, "breakers", t.breakerId);
  change(app, `found ${plural(ids.length, "device")} in ${group.roomName} on ${breakerTitle(breaker)}`, (p) => { for (const id of ids) markDead(p, t, id); });
  toast(`Marked ${plural(ids.length, "device")}.`, { text: "Undo", run: () => { app.undo(); app.saveTest(); } });
}

// ---- "I've switched it back on": from the bar on any screen, or the foot of the marking page

export function backOn(app) {
  const t = app.test;
  if (!t || !m.find(app.project, "breakers", t.breakerId)) {
    drop(app);
    app.render();
    return;
  }
  if (app.mode) app.cancelMode();
  t.off = false;
  reconcile(app.project, t);
  const left = stillAssigned(app.project, t);
  if (left.guessed.length + left.tested.length) t.review = true;
  else finish(app);
  app.saveTest();
  Object.assign(state(app), { page: "test", breakerId: t.breakerId });
  app.tab = "panel";
  app.render();
}

// The app was opened with a breaker still marked off.
export function askBackOn(app) {
  const breaker = m.find(app.project, "breakers", app.test.breakerId);
  ask(`Is breaker ${breakerName(breaker)} back on?`, "", [
    { text: "Yes, it's back on", cls: "main", run: () => backOn(app) },
    { text: "No, not yet" },
  ]);
}

// Record the breaker as tested (even if nothing was found on it) and show the summary.
function finish(app) {
  const t = app.test, breaker = m.find(app.project, "breakers", t.breakerId);
  t.review = false;
  t.done = true;
  // The summary is worked out inside the change, because the change redraws the page, which shows it.
  app.commit(`tested ${breakerTitle(breaker)}`, (p) => {
    m.markSwept(p, breaker.id);
    t.summary = finishSummary(p, t);
  });
}

// ---- stage 3: still assigned here

function reviewPage(app, breaker, top) {
  const t = app.test, left = stillAssigned(app.project, t);
  const next = () => {
    const now = stillAssigned(app.project, t);
    if (!now.guessed.length && !now.tested.length) finish(app);
    redraw(app);
  };
  const takeOff = (list) => {
    change(app, `took ${joinLabels(list.map((d) => d.label))} off ${breakerTitle(breaker)}`, (p) => { for (const d of list) m.setBreaker(p, d.id, "", false); });
    next();
  };
  const group = (title, list) => list.length > 0 && [
    h("h3", {}, title),
    list.length > 1 && button("Take them all off this breaker", () => takeOff(list)),
    h("ul", { class: "rows" }, list.map((d) => h("li", {},
      h("span", {}, `${d.label}, ${displayName(d)}`),
      h("div", { class: "buttons" },
        button("Take it off", () => takeOff([d]), "main"),
        button("Keep it", () => { t.kept.push(d.id); next(); }))))),
  ];
  return h("section", { class: "page" }, top,
    // Nothing left (the app was closed part way through this step): just carry on.
    h("p", {}, left.text || "Nothing else is left on this breaker to check."),
    h("p", { class: "small" }, "A wrong guess, or a circuit that has been changed, shows up here. Only floors you ticked as checked are counted."),
    group("Guess", left.guessed),
    group("Found by an earlier test", left.tested),
    button(left.text ? "Keep all" : "Continue", () => { finish(app); redraw(app); }));
}

// ---- stage 4: the summary

    next

      : h("p", {}, "Every breaker has been tested."));
}
