# Circuit Map Plan 1: Core Data Modules Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build and test the DOM-free core of Circuit Map: the fixture catalog, project data model, device labels, merge-on-import, project file format and CSV export.

**Architecture:** Plain JavaScript ES modules in `src/`, no build step, no dependencies. A project is one plain JSON-safe object; every module here is a set of pure functions over it, so the browser UI (Plans 2-4) and the tests share the same code. Functions that stamp a time take an optional `now` argument so tests are deterministic.

**Tech Stack:** JavaScript (ES modules), Node 22 built-in test runner (`node --test`), `node:assert/strict`.

**Spec:** `docs/superpowers/specs/2026-10-06-circuit-map-app-design.md`

```js
// Device labels look like KIT-07: a room code, a dash, a number. Labels are permanent once given.

export function labelNumber(label) {
  const m = /-(\d+)$/.exec(label);
  return m ? Number(m[1]) : 0;
}

export function nextLabel(project, code) {
  let max = 0;
  for (const d of project.devices) {
    if (d.label.startsWith(code + "-")) max = Math.max(max, labelNumber(d.label));
  }
  return code + "-" + String(max + 1).padStart(2, "0");
}

// A room code is a capital letter followed by 1-3 capitals or digits.
export const CODE_PATTERN = /^[A-Z][A-Z0-9]{1,3}$/;

export function suggestCode(name, takenCodes = []) {
  const words = name.toUpperCase().replace(/[^A-Z0-9 ]/g, "").split(/\s+/).filter(Boolean);
  let base = words.length > 1 ? words.map((w) => w[0]).join("") : words[0] || "RM";
  if (!/^[A-Z]/.test(base)) base = "R" + base;
  base = base.slice(0, 3);
  while (base.length < 2) base += "X";
  if (!takenCodes.includes(base)) return base;
  for (let n = 2; n < 100; n++) {
    const code = base.slice(0, 4 - String(n).length) + n;
    if (!takenCodes.includes(code)) return code;
  }
  throw new Error("no free room code for " + name);
}

// After a merge two devices can share a label. The one created first keeps it; later ones move to the
// next free number. Returns [{id, from, to}] for the summary shown to the user.
export function fixLabelCollisions(project) {
  const order = [...project.devices].sort((a, b) => a.created - b.created || (a.id < b.id ? -1 : 1));
  const seen = new Set();
  const dupes = [];
  for (const d of order) {
    if (seen.has(d.label)) dupes.push(d);
    else seen.add(d.label);
  }
  const changes = [];
  for (const d of dupes) {
    const to = nextLabel(project, d.label.replace(/-\d+$/, ""));
    changes.push({ id: d.id, from: d.label, to });
    d.label = to;
  }
  return changes;
}
```

```js
// The project: one plain object holding everything about one house. All edits go through this module.
import { CODE_PATTERN } from "./ids.js";

export const FORMAT_VERSION = 1;
export const COLLECTIONS = ["floors", "rooms", "devices", "links", "panels", "breakers"];

export function newId() {
  return globalThis.crypto.randomUUID();
}

export function createProject(name, now = Date.now()) {
  return { formatVersion: FORMAT_VERSION, id: newId(), name, updated: now,
    floors: [], rooms: [], devices: [], links: [], panels: [], breakers: [], deleted: [] };
}

export function rename(project, name, now = Date.now()) {
  project.name = name;
  project.updated = now;
}

export function find(project, coll, id) {
  return project[coll].find((r) => r.id === id) || null;
}

export function add(project, coll, fields, now = Date.now()) {
  const rec = { id: newId(), ...fields, updated: now };
  project[coll].push(rec);
  return rec;
}

export function update(project, coll, id, fields, now = Date.now()) {
  const rec = find(project, coll, id);
  if (!rec) throw new Error(`no ${coll} record ${id}`);
  Object.assign(rec, fields, { updated: now });
  return rec;
}

// ---- floors

export function addFloor(project, name, now = Date.now()) {
  return add(project, "floors", { name, order: project.floors.length, imageId: "", pxPerIn: 0 }, now);
}

export function scaleFromPoints(a, b, inches) {
  const px = Math.hypot(b[0] - a[0], b[1] - a[1]);
  if (!px) throw new Error("pick two different points");
  if (!(inches > 0)) throw new Error("enter the distance between the points");
  return px / inches;
}

// ---- rooms

export function addRoom(project, { floorId, name, code }, now = Date.now()) {
  if (!CODE_PATTERN.test(code)) throw new Error("room code must be 2-4 capitals or digits, starting with a letter");
  if (project.rooms.some((r) => r.code === code)) throw new Error(`room code ${code} is already used`);
  return add(project, "rooms", { floorId, code, name, outline: [] }, now);
}

export function pointInOutline(outline, x, y) {
  if (outline.length < 3) return false;
  let inside = false;
  for (let i = 0, j = outline.length - 1; i < outline.length; j = i++) {
    const [xi, yi] = outline[i], [xj, yj] = outline[j];
    if (yi > y !== yj > y && x < ((xj - xi) * (y - yi)) / (yj - yi) + xi) inside = !inside;
  }
  return inside;
}

export function roomsAt(project, floorId, x, y) {
  return project.rooms.filter((r) => r.floorId === floorId && pointInOutline(r.outline, x, y));
}
```

```js
// so a later merge does not bring them back.

const CHILDREN = {
  floors: (p, id) => [...p.rooms.filter((r) => r.floorId === id).map((r) => ["rooms", r.id]),
    ...p.devices.filter((d) => d.floorId === id).map((d) => ["devices", d.id])],
  rooms: (p, id) => p.devices.filter((d) => d.roomId === id).map((d) => ["devices", d.id]),
  devices: (p, id) => p.links.filter((l) => l.switchId === id || l.loadId === id).map((l) => ["links", l.id]),
  panels: (p, id) => p.breakers.filter((b) => b.panelId === id).map((b) => ["breakers", b.id]),
  breakers: () => [],
  links: () => [],
};

function cascade(project, coll, id, out = []) {
  if (out.some(([c, i]) => c === coll && i === id)) return out;
  out.push([coll, id]);
  for (const [c, i] of CHILDREN[coll](project, id)) cascade(project, c, i, out);
  return out;
}

// What else goes if this is deleted: counts per collection, plus `unassigned` = devices that lose their breaker.
export function affected(project, coll, id) {
  const list = cascade(project, coll, id);
  const counts = {};
  for (const [c] of list.slice(1)) counts[c] = (counts[c] || 0) + 1;
  const ids = new Set(list.map(([, i]) => i));
  const unassigned = project.devices.filter((d) => ids.has(d.breakerId)).length;
  if (unassigned) counts.unassigned = unassigned;
  return counts;
}

export function remove(project, coll, id, now = Date.now()) {
  const ids = new Set(cascade(project, coll, id).map(([, i]) => i));
  for (const c of COLLECTIONS) project[c] = project[c].filter((r) => !ids.has(r.id));
  for (const i of ids) project.deleted.push({ id: i, at: now });
  for (const d of project.devices) {
    if (ids.has(d.breakerId)) Object.assign(d, { breakerId: "", tested: false, updated: now });
  }
  for (const p of project.panels) {
    if (ids.has(p.fedFromBreakerId)) Object.assign(p, { fedFromBreakerId: "", updated: now });
    if (ids.has(p.deviceId)) Object.assign(p, { deviceId: "", updated: now });
  }
}
```

```js
// Combine two copies of the same project (this device's and an imported file's) without losing work.
import { COLLECTIONS } from "./model.js";
import { fixLabelCollisions } from "./ids.js";

// Key order must not matter when comparing two copies of a record. Records are flat except for arrays.
const canon = (rec) => JSON.stringify(rec, Object.keys(rec).sort());

// Later change wins. On a tie any rule will do, as long as both devices pick the same one.
function newer(x, y) {
  if (x.updated !== y.updated) return x.updated > y.updated ? x : y;
  return canon(x) >= canon(y) ? x : y;
}

const byId = (a, b) => (a.id < b.id ? -1 : a.id > b.id ? 1 : 0);

// Remove or clear anything that points at a record that no longer exists.
function tidy(p) {
  const ids = (coll) => new Set(p[coll].map((r) => r.id));
  const floors = ids("floors"), panels = ids("panels");
  p.rooms = p.rooms.filter((r) => floors.has(r.floorId));
  p.devices = p.devices.filter((d) => floors.has(d.floorId));
  p.breakers = p.breakers.filter((b) => panels.has(b.panelId));
  const rooms = ids("rooms"), breakers = ids("breakers"), devices = ids("devices");
  for (const d of p.devices) {
    if (d.roomId && !rooms.has(d.roomId)) d.roomId = "";
    if (d.breakerId && !breakers.has(d.breakerId)) Object.assign(d, { breakerId: "", tested: false });
  }
  for (const panel of p.panels) {
    if (panel.fedFromBreakerId && !breakers.has(panel.fedFromBreakerId)) panel.fedFromBreakerId = "";
    if (panel.deviceId && !devices.has(panel.deviceId)) panel.deviceId = "";
  }
  const seen = new Set();
  p.links = p.links.filter((l) => {
    const key = [l.switchId, l.toggle, l.loadId, l.loadPart].join("|");
    if (!devices.has(l.switchId) || !devices.has(l.loadId) || seen.has(key)) return false;
    seen.add(key);
    return true;
  });
}

export function merge(local, incoming) {
  if (local.id !== incoming.id) {
    const err = new Error("This file is for a different house.");
    err.code = "different-project";
    throw err;
  }
  const gone = new Map();
  for (const t of [...local.deleted, ...incoming.deleted]) gone.set(t.id, Math.max(t.at, gone.get(t.id) || 0));

  const head = newer({ name: local.name, updated: local.updated }, { name: incoming.name, updated: incoming.updated });
  const out = { formatVersion: Math.max(local.formatVersion, incoming.formatVersion), id: local.id,
    name: head.name, updated: head.updated };
  for (const coll of COLLECTIONS) {
    const all = new Map(local[coll].map((r) => [r.id, r]));
    for (const r of incoming[coll]) all.set(r.id, all.has(r.id) ? newer(all.get(r.id), r) : r);
    out[coll] = [...all.values()].filter((r) => !(gone.get(r.id) >= r.updated)).map((r) => structuredClone(r)).sort(byId);
  }
  out.deleted = [...gone].map(([id, at]) => ({ id, at })).sort(byId);
  tidy(out);

  const summary = { added: 0, updated: 0, removed: 0, renumbered: fixLabelCollisions(out) };
  for (const coll of COLLECTIONS) {
    const mine = new Map(local[coll].map((r) => [r.id, r]));
    const kept = new Set();
    for (const r of out[coll]) {
      kept.add(r.id);
      if (!mine.has(r.id)) summary.added++;
      else if (canon(mine.get(r.id)) !== canon(r)) summary.updated++;
    }
    for (const id of mine.keys()) if (!kept.has(id)) summary.removed++;
  }
  return { project: out, summary };
}
```

```js
// The project file that moves between devices: JSON text holding the project and its plan images.
import { FORMAT_VERSION, COLLECTIONS } from "./model.js";

export const EXTENSION = ".circuitmap";

export function fileName(project, date = new Date()) {
  const p = (n) => String(n).padStart(2, "0");
  const stamp = `${date.getFullYear()}-${p(date.getMonth() + 1)}-${p(date.getDate())} ${p(date.getHours())}${p(date.getMinutes())}`;
  const safe = project.name.replace(/[\\/:*?"<>|]/g, "").trim() || "House";
  return `${safe} ${stamp}${EXTENSION}`;
}

export function serialize(project, images = {}) {
  return JSON.stringify({ app: "circuit-map", project, images });
}

function importError(code, message) {
  const err = new Error(message);
  err.code = code;
  return err;
}

// Returns { project, images } or throws an Error whose .code is "unreadable", "format" or "newer"
// and whose .message can be shown to the user as is.
export function parse(text) {
  let data;
  try {
    data = JSON.parse(text);
  } catch {
    throw importError("unreadable", "This file could not be read.");
  }
  const project = data && data.app === "circuit-map" ? data.project : null;
  if (!project || typeof project.id !== "string") throw importError("format", "This is not a Circuit Map project file.");
  if (project.formatVersion > FORMAT_VERSION) {
    throw importError("newer", "This file was made by a newer version of Circuit Map. Reload the app to update it, then try again.");
  }
  for (const coll of [...COLLECTIONS, "deleted"]) {
    if (!Array.isArray(project[coll])) throw importError("format", "This is not a Circuit Map project file.");
  }
  return { project, images: data.images && typeof data.images === "object" ? data.images : {} };
}
```

```js
// The device list as CSV text, for a spreadsheet or a printout.
import { displayName } from "./catalog.js";
import { find, breakerKey } from "./model.js";

function cell(value) {
  let s = value == null ? "" : String(value);
  if (/^[=+\-@]/.test(s)) s = "'" + s; // stop a spreadsheet treating typed text as a formula
  return /[",\r\n]/.test(s) ? '"' + s.replace(/"/g, '""') + '"' : s;
}

export function devicesCsv(project) {
  const label = (id) => (find(project, "devices", id) || { label: "" }).label;
  const rows = [["label", "floor", "room", "type", "height_in", "panel", "breaker", "amps", "tested",
    "controls", "switched_by", "note"]];
  const devices = [...project.devices].sort((a, b) => a.label.localeCompare(b.label, "en", { numeric: true }));
  for (const d of devices) {
    const floor = find(project, "floors", d.floorId);
    const room = find(project, "rooms", d.roomId);
    const breaker = find(project, "breakers", d.breakerId);
    const panel = breaker && find(project, "panels", breaker.panelId);
    rows.push([
      d.label, floor ? floor.name : "", room ? room.name : "", displayName(d), d.heightIn,
      panel ? panel.name : "", breaker ? breakerKey(breaker) : "", breaker ? breaker.amps : "",
      breaker ? (d.tested ? "yes" : "guess") : "",
      project.links.filter((l) => l.switchId === d.id).map((l) => label(l.loadId) + (l.loadPart ? " " + l.loadPart : "")).join(" "),
      project.links.filter((l) => l.loadId === d.id).map((l) => label(l.switchId)).join(" "),
      d.note,
    ]);
  }
  return rows.map((r) => r.map(cell).join(",")).join("\r\n") + "\r\n";
}
```
